# Software Passport Registry - crawler policy
#
# Served as a real static file from the Vite public/ directory. Vercel
# resolves static files before the SPA catch-all rewrite, so this is
# returned as text/plain rather than the index.html shell.
#
# robots.txt is NOT a security boundary. Every route listed under Disallow
# is additionally protected server-side by Firebase authentication,
# role checks and tenant-scoped queries; this file only keeps crawlers off
# pages that have no public value.

User-agent: *

# Public marketing and self-service surfaces.

# Authenticated application. These render tenant-scoped customer data and
# are behind auth; they must never appear in search results.
# Note /registry is an ALIAS of the authenticated /passports view, not a
# public registry - it is deliberately disallowed.

# Sign-in is public but has no search value.
Disallow: /login

# Free Review result links embed a signed, time-boxed bearer token. They are
# publicly reachable by design so a visitor can return to or share their own
# result, but they must never be crawled, cached or indexed.
Disallow: /free-review/result

# APIs, including signed public share links. A Passport verification link
# is a time-boxed bearer token; it must not be crawled, cached or indexed.
Disallow: /api/
